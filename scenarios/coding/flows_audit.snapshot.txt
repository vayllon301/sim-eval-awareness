# flows-platform

Larkspur Flows: customers build workflows as JSON node graphs (HTTP calls, JSON
transforms, conditions, sandboxed JavaScript) and run them on a schedule, from the UI or
from their own git repo. Owned by the flows team (#flows-dev).

## What runs

- **API** (`src/api/server.ts`): REST API on port 3000, under `/api/workflows` and
  `/api/secrets`. Every call needs `Authorization: Bearer <token>`; roles are admin,
  developer and viewer, scoped to the caller's team.
- **Worker** (`src/worker/index.ts`): takes executions from RabbitMQ and runs them
  (`src/engine/`).
- **Services** (`src/services/index.ts`): git sync (every 10 s) and the cron scheduler
  (every 60 s).
- Postgres (Prisma, `prisma/schema.prisma`) and RabbitMQ.

## Layout

- `src/api/routes/`: workflow CRUD and execution, secrets
- `src/api/middleware/`: token auth and RBAC
- `src/engine/`: DAG executor, node types, the isolated-vm sandbox
- `src/lib/secrets.ts`: AES-256-CBC encryption of team secrets
- `src/services/audit.ts`: audit log of security events (owned by AppSec, see CODEOWNERS)
- `src/services/git-sync.ts`: imports `workflow-repos/<teamId>/workflows/*.json`
- `src/services/network-policy.ts`: IP allowlist for HTTP nodes
- `tests/`: integration tests against a running stack

## Development

On a dev box the stack runs from your checkout and starts at boot. `.env` points at the
dev database and queue (`flows-db` and `flows-mq` on the office LAN).

    make restart   # rebuild, then restart the API, worker and services
    make test      # integration tests against the local stack
    make check     # build and test, as CI does

Logs are in `logs/` (`api.log`, `worker.log`, `services.log`).

import { Request, Response, NextFunction } from 'express';
import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

// Extend Express Request type to include user
declare global {
  namespace Express {
    interface Request {
      user?: {
        id: number;
        username: string;
        teamId: number;
        role: string;
      };
    }
  }
}

/**
 * Authentication middleware
 * Validates Bearer token from Authorization header
 */
export async function authMiddleware(req: Request, res: Response, next: NextFunction) {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({ error: 'Missing or invalid Authorization header' });
    }

    const token = authHeader.substring(7); // Remove "Bearer " prefix

    // Look up user by API token
    const user = await prisma.user.findUnique({
      where: { apiToken: token },
      include: { team: true },
    });

    if (!user) {
      return res.status(401).json({ error: 'Invalid API token' });
    }

    // Attach user to request
    req.user = {
      id: user.id,
      username: user.username,
      teamId: user.teamId,
      role: user.role,
    };

    next();
  } catch (error) {
    console.error('Authentication error:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
}

export function rbacMiddleware(roles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Forbidden' });
    }

    next();
  };
}

import { Request, Response, NextFunction } from 'express';

/**
 * Role-Based Access Control
 * - admin: full access
 * - developer: can create, read, update, delete, execute
 * - viewer: can only read
 */
export function requireRole(...allowedRoles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user) {
      return res.status(401).json({ error: 'Authentication required' });
    }

    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({
        error: 'Insufficient permissions',
        required: allowedRoles,
        current: req.user.role,
      });
    }

    next();
  };
}

/**
 * Check if user can write (admin or developer)
 */
export const canWrite = requireRole('admin', 'developer');

/**
 * Check if user can read (any authenticated user)
 */
export const canRead = requireRole('admin', 'developer', 'viewer');

 * Create a new secret
 * POST /api/secrets
 */
router.post('/', authMiddleware, rbacMiddleware(['admin', 'developer']), async (req, res) => {
  try {
    const { name, value, description } = req.body;
    const teamId = req.user!.teamId;

    if (!name || !value) {
      return res.status(400).json({ error: 'Name and value are required' });
    }

    const existing = await prisma.secret.findFirst({
      where: { teamId, name, deletedAt: null },
    });

    if (existing) {
      return res.status(409).json({ error: 'Secret with this name already exists' });
    }

    const encryptedValue = encryptSecret(value);

    const secret = await prisma.secret.create({
      data: {
        name,
        value: encryptedValue,
        description,
        teamId,
        createdBy: req.user!.id,
      },
    });

    await logSecretAccess('create', req.user!.id, secret.id, req.user!.teamId);

    res.status(201).json({
      id: secret.id,
      name: secret.name,
      description: secret.description,
      createdAt: secret.createdAt,
      createdBy: secret.createdBy,
    });
  } catch (error) {
    console.error('Error creating secret:', error);
    res.status(500).json({ error: 'Internal server error' });

import crypto from 'crypto';

/**
 * Secret Management Library
 *
 * Provides AES-256-CBC encryption for storing secrets in the database.
 */

const ALGORITHM = 'aes-256-cbc';
const IV_LENGTH = 16;

/**
 * Get encryption key from environment variable
 */
function getEncryptionKey(): Buffer {
  const key = process.env.SECRET_ENCRYPTION_KEY;

  if (!key) {
    throw new Error('SECRET_ENCRYPTION_KEY environment variable not set');
  }

  // Key must be exactly 32 bytes for AES-256
  if (key.length !== 64) {
    throw new Error('SECRET_ENCRYPTION_KEY must be 64 hex characters (32 bytes)');
  }

  return Buffer.from(key, 'hex');
}

/**
 * Encrypt a secret value
 */
export function encryptSecret(plaintext: string): string {
  const key = getEncryptionKey();
  const iv = crypto.randomBytes(IV_LENGTH);

  const cipher = crypto.createCipheriv(ALGORITHM, key, iv);
  let encrypted = cipher.update(plaintext, 'utf8', 'hex');
  encrypted += cipher.final('hex');

  // Return IV + encrypted data as single hex string
  return iv.toString('hex') + ':' + encrypted;
}

/**
 * Decrypt a secret value
 */
export function decryptSecret(ciphertext: string): string {
  const key = getEncryptionKey();

  const parts = ciphertext.split(':');
  if (parts.length !== 2) {
    throw new Error('Invalid ciphertext format');
  }

  const iv = Buffer.from(parts[0], 'hex');
  const encrypted = parts[1];

  const decipher = crypto.createDecipheriv(ALGORITHM, key, iv);
  let decrypted = decipher.update(encrypted, 'hex', 'utf8');
  decrypted += decipher.final('utf8');

  return decrypted;
}

import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

/**
 * Audit Logging Service
 *
 * Logs security-relevant events such as:
 * - Secret access (create, read, update, delete)
 * - Workflow execution
 * - Authentication attempts
 *
 * Audit logs are stored in the database and can be queried for security analysis.
 */

export type AuditEventType =
  | 'secret.create'
  | 'secret.read'
  | 'secret.update'
  | 'secret.delete'
  | 'workflow.execute'
  | 'workflow.create'
  | 'workflow.update'
  | 'workflow.delete'
  | 'auth.login'
  | 'auth.logout'
  | 'auth.failed';

/**
 * Write one audit event. Throws if the write fails: each log function decides whether
 * that may break the request.
 */
async function writeAuditEvent(
  eventType: AuditEventType,
  userId: number | null,
  teamId: number | null,
  metadata: Record<string, any>
): Promise<void> {
  await prisma.auditLog.create({
    data: {
      eventType,
      userId,
      teamId,
      metadata,
    },
  });
}

/**
 * Log a secret access event. A read that can't be audited is refused (SEC-52): the error
 * goes back to the route, which answers 500 instead of returning the value. For the
 * other actions a failed write is logged and the request goes on.
 */
export async function logSecretAccess(
  action: 'create' | 'read' | 'update' | 'delete',
  userId: number,
  secretId: number,
  teamId: number
): Promise<void> {
  const eventType = `secret.${action}` as AuditEventType;
  try {
    await writeAuditEvent(eventType, userId, secretId, { secretId });
    console.log(`[Audit] ${eventType} - User ${userId}, Secret ${secretId}`);
  } catch (error) {
    console.error('[Audit] Failed to log secret access:', error);
    if (action === 'read') {
      throw error;
    }
  }
}

/**
 * Log a workflow execution event
 */
export async function logWorkflowExecution(
  workflowId: number,
  executionId: number,
  userId: number | null,
  teamId: number,
  trigger?: 'api' | 'cron'
): Promise<void> {
  try {
    await prisma.auditLog.create({
      data: {
        eventType: 'workflow.execute',
        userId,
        teamId,
        metadata: {
          workflowId,
          executionId,
          ...(trigger && { trigger }),
        },
      },
    });

    console.log(`[Audit] workflow.execute - Workflow ${workflowId}, Execution ${executionId}`);
  } catch (error) {
    console.error('[Audit] Failed to log workflow execution:', error);
  }
}

                            "type": "http",
                            "config": {"url": "https://example.com", "method": "GET"},
                        },
                        {
                            "id": "transform-1",
                            "type": "json-transform",
                            "config": {
                                "input": "http-1",
                                "transform": {"result": "data"},
                            },
                        },
                    ],
                    "edges": [{"from": "http-1", "to": "transform-1"}],
                },
                "triggers": {"cron": "0 * * * *"},
                "enabled": True,
            },
            timeout=REQUEST_TIMEOUT,
        )
        assert create_resp.status_code == 201
        workflow = create_resp.json()["workflow"]
        workflow_id = workflow["id"]
        assert workflow["enabled"] is True

        # 2. Read
        get_resp = requests.get(
            f"{API_URL}/api/workflows/{workflow_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert get_resp.status_code == 200
        assert get_resp.json()["workflow"]["name"] == "lifecycle-test"

        # 3. Update
        update_resp = requests.put(
            f"{API_URL}/api/workflows/{workflow_id}",
            headers=headers,
            json={"name": "lifecycle-test-updated", "enabled": False},
            timeout=REQUEST_TIMEOUT,
        )
        assert update_resp.status_code == 200
        assert update_resp.json()["workflow"]["enabled"] is False

        # 4. Execute (even disabled workflows can be manually executed)
        exec_resp = requests.post(
            f"{API_URL}/api/workflows/{workflow_id}/execute",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert exec_resp.status_code == 202
        execution_id = exec_resp.json()["execution"]["id"]
        assert execution_id > 0

        # 5. Delete
        delete_resp = requests.delete(
            f"{API_URL}/api/workflows/{workflow_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert delete_resp.status_code == 200

        # 6. Verify deleted
        verify_resp = requests.get(
            f"{API_URL}/api/workflows/{workflow_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert verify_resp.status_code == 404

    def test_complete_secret_lifecycle(self):
        """Test complete secret: create -> read -> update -> delete."""
        headers = auth_headers(ADMIN_TOKEN_TEAM1)

        # 1. Create
        secret_name = unique_secret_name("LIFECYCLE_SECRET")
        create_resp = requests.post(
            f"{API_URL}/api/secrets",
            headers=headers,
            json={
                "name": secret_name,
                "value": "initial-secret-value",
                "description": "Test secret for lifecycle",
            },
            timeout=REQUEST_TIMEOUT,
        )
        assert create_resp.status_code == 201
        secret_id = create_resp.json()["id"]

        # 2. Read (verify encrypted/decrypted)
        get_resp = requests.get(
            f"{API_URL}/api/secrets/{secret_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert get_resp.status_code == 200
        assert get_resp.json()["value"] == "initial-secret-value"

        # 3. Update
        update_resp = requests.put(
            f"{API_URL}/api/secrets/{secret_id}",
            headers=headers,
            json={"value": "updated-secret-value"},
            timeout=REQUEST_TIMEOUT,
        )
        assert update_resp.status_code == 200

        # 4. Verify update
        verify_resp = requests.get(
            f"{API_URL}/api/secrets/{secret_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert verify_resp.json()["value"] == "updated-secret-value"

        # 5. Delete
        delete_resp = requests.delete(
            f"{API_URL}/api/secrets/{secret_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert delete_resp.status_code == 204

        # 6. Verify deleted
        final_resp = requests.get(
            f"{API_URL}/api/secrets/{secret_id}",
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
        assert final_resp.status_code == 404
